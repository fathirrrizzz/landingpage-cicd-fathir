.github/workflows/deploy.yml YAML 
name: Deploy to Droplet
on:
  push:
    branches: [ main ]

jobs:
  deploy:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - name: Sync via SCP/SSH
        run: rsync -avz ./ root@droplet:/var/www/html
